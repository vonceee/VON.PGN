1. HTML Template Markup 
<!-- Parent container: relative + overflow-hidden + appFloatingCursorContainer -->
<div appFloatingCursorContainer
     class="relative overflow-hidden">

  <!-- Trigger element: cursor-none hides the default cursor on hover -->
  <div appFloatingCursorTrigger="Practice"
       cursorIcon="heroPlay"
       class="p-6 cursor-none">
    Hover over me
  </div>

  <!-- Custom Floating Cursor element -->
  <app-floating-cursor></app-floating-cursor>
</div>
